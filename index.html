function renderAmbulances(list) { const grid = document.getElementById("amb-grid"); const countEl = document.getElementById("amb-count"); grid.innerHTML = ""; countEl.textContent = `Showing ${list.length} ambulances`; if (list.length === 0) { grid.innerHTML = `
No ambulance services match your search.
`; return; } list.forEach(item => { const hasPhone = item.phone && item.phone !== "N/A"; const phoneLink = hasPhone ? `tel:${item.phone}` : "#"; const isFree = item.pricing.toLowerCase() === "free"; const card = document.createElement("div"); card.className = "bg-white p-4 rounded-2xl border border-slate-200/80 shadow-sm flex flex-col justify-between hover:border-slate-300 transition"; card.innerHTML = `
${item.category} • ${item.service_type}

${item.name}

${isFree ? '100% Free' : 'Paid'} 24x7

${item.address || 'Moradabad'}

${item.features.oxygen ? '✓' : '✕'} Oxygen Support ${item.features.icu ? '✓' : '✕'} ICU / Cardiac ${item.features.als ? '✓' : '✕'} ALS Facility ${item.features.intercity ? '✓' : '✕'} Intercity / NCR
📞 ${hasPhone ? `Call Ambulance (${item.phone})` : 'Phone Unavailable'} `; grid.appendChild(card); }); }
